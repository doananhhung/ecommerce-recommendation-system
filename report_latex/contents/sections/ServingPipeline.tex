\chapter{Triển khai Hệ thống Phục vụ (Serving Pipeline)}

\section{Thiết kế luồng phục vụ hệ thống (System Flow)}
Serving Pipeline là cầu nối trực tiếp tiếp nhận các yêu cầu truy vấn của người dùng thời gian thực thông qua API và trả về kết quả khuyến nghị cá nhân hóa. Chúng em xây dựng máy chủ FastAPI và Uvicorn hiệu năng cao:
\begin{itemize}
    \item \textbf{In-Memory Feature Tables:} Tải toàn bộ thuộc tính tĩnh Parquet (\texttt{user\_features.parquet}, \texttt{item\_features.parquet}) và FAISS Index lên bộ nhớ RAM ngay khi API khởi động, triệt tiêu hoàn toàn nghẽn Disk I/O.
    \item \textbf{Luồng vận hành Serving:} 
        \begin{enumerate}
            \item Tiếp nhận yêu cầu \texttt{GET /recommend/\{user\_id\}?session\_items=A,B,C} chứa ID người dùng và lịch sử các sản phẩm vừa click tương tác trong phiên.
            \item Phân phối song song hai kênh triệu hồi FAISS (Sở thích lâu dài \& Phiên tức thì).
            \item Gộp ứng viên và loại trùng lặp (loại bỏ các mặt hàng trùng nhau).
            \item Trích xuất thuộc tính tĩnh từ RAM Feature Store, ghép nối với đặc trưng phiên tính động tại thời điểm API chạy.
            \item Đưa dữ liệu bảng vào LightGBM Ranker để chấm điểm xác suất chuyển đổi CTR kháng lỗi.
            \item Sắp xếp điểm số giảm dần và trả về Top 20 sản phẩm khuyến nghị tốt nhất định dạng JSON.
        \end{enumerate}
\end{itemize}

\section{Truy xuất vector lân cận gần nhất siêu tốc bằng FAISS Index}
Để đảm bảo khâu triệu hồi thô diễn ra trong chưa đầy 3ms, chúng em sử dụng thư viện \textbf{FAISS (Facebook AI Similarity Search)} viết bằng C++:
\begin{itemize}
    \item Dựng chỉ mục tìm kiếm vector xấp xỉ Flat Inner Product (\texttt{faiss.IndexFlatIP}) dựa trên tích vô hướng Dot Product của embeddings ẩn 64 chiều.
    \item Tìm kiếm song song: Truy vấn cùng lúc 100 sản phẩm lân cận cho kênh lâu dài và 100 sản phẩm lân cận cho kênh phiên tức thì (thông qua Average Session Embedding).
    \item FAISS cho phép thực hiện tìm kiếm vector trên RAM chỉ tốn \textbf{0.013 giây} cho cơ sở dữ liệu \textbf{24,428 sản phẩm} lúc khởi động, đảm bảo thời gian quét vector phục vụ API chỉ mất dưới \textbf{2.24ms}.
\end{itemize}

\section{Tích hợp Pipeline và cơ chế Ranker thời gian thực kháng lỗi}
Khâu xếp hạng lại trực tuyến LightGBM GBDT được thiết lập cơ chế \textbf{kháng lỗi suy luận động (Fault-Tolerant Serving)} vô cùng mạnh mẽ tại \texttt{src/serving/ranker.py}:
\begin{itemize}
    \item \textbf{Dynamic Feature Alignment (Tự động so khớp đặc trưng):} Trong môi trường thực tế, nếu mô hình LightGBM được huấn luyện với các cột đặc trưng bổ sung nhưng Feature Store Serving bị thiếu cột, LightGBM sẽ crash. Hệ thống tự động gọi \texttt{self.model.feature\_name()} lúc khởi động để trích xuất danh sách cột mô hình mong đợi. Sau đó, nó tự động đối chiếu, bù đắp các cột thiếu bằng giá trị mặc định (\texttt{\textless UNKNOWN\textgreater} cho phân loại, \texttt{0.0} cho số học) và sắp xếp lại các cột theo đúng thứ tự mô hình yêu cầu trước khi chấm điểm. Điều này đảm bảo API không bao giờ bị gián đoạn hay crash.
    \item \textbf{Cold-Start Fallback:} Nếu gặp một \texttt{user\_id} mới hoàn toàn (chưa có trong Feature Store), hệ thống tự động chuyển hướng gợi ý danh sách sản phẩm thịnh hành nhất (\texttt{\_popular\_fallback\_recommendations}) dựa trên tương tác lịch sử thực tế mà không gây lỗi runtime.
\end{itemize}

\section{Kiểm thử hệ thống (Smoke Testing)}
Để xác thực tính đúng đắn và hiệu năng thực tế của luồng phục vụ, chúng em phát triển kịch bản kiểm thử tích hợp tự động \textbf{Smoke Testing} (\texttt{tests/smoke\_test.py}):
\begin{enumerate}
    \item Khởi chạy tự động Pipeline, nạp encoders, nạp PyTorch MF model và LightGBM ranker.
    \item Kiểm tra luồng khởi động: Pipeline sẵn sàng phục vụ chỉ sau \textbf{0.85 giây}.
    \item Mô phỏng truy vấn Serving cho User ID $244951053$. Kết quả suy luận thành công mượt mà, trả về đúng định dạng cấu trúc JSON chứa Top 5 sản phẩm đề xuất tốt nhất.
    \item Tổng độ trễ Serving được đo lường thực tế đạt mức rất thấp: \textbf{16.89 ms} (nằm sâu dưới giới hạn tiêu chuẩn công nghiệp 50ms).
\end{enumerate}
